function doGet(e) { const orderId = String(e.parameter.orderId || "").trim(); if (!orderId) { return ContentService.createTextOutput(JSON.stringify({ status: "請提供訂單號" })).setMimeType(ContentService.MimeType.JSON); } // ✅ Cache（60秒） const cache = CacheService.getScriptCache(); const cached = cache.get(orderId); if (cached) { return ContentService .createTextOutput(cached) .setMimeType(ContentService.MimeType.JSON); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1"); // ✅ 只抓必要欄位（A~H = 8欄） const lastRow = sheet.getLastRow(); if (lastRow < 2) { return ContentService.createTextOutput(JSON.stringify({ status: "尚無資料" })).setMimeType(ContentService.MimeType.JSON); } const data = sheet.getRange(2, 1, lastRow - 1, 8).getValues(); let items = []; let status = ""; let time = ""; for (let i = 0; i < data.length; i++) { const row = data[i]; const sheetOrderId = String(row[1]).trim(); if (sheetOrderId === orderId) { items.push({ name: row[4], qty: row[5], price: row[6], image: row[7] }); if (!status) status = String(row[2]).trim(); if (!time) time = row[3]; } } let response; if (items.length > 0) { response = { status: status, time: time, items: items }; } else { response = { status: "找不到訂單" }; } const json = JSON.stringify(response); // ✅ 存 cache cache.put(orderId, json, 60); return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1"); const data = JSON.parse(e.postData.contents); const email = data.email || "無email"; const orderId = data.name ? data.name.replace("#", "") : "無訂單號"; const createdAt = data.created_at ? new Date(data.created_at).toLocaleString() : "無時間"; const items = data.line_items || []; // ✅ 防重複 const lastRow = sheet.getLastRow(); if (lastRow > 1) { const existingIds = sheet.getRange(2, 2, lastRow - 1, 1) .getValues() .flat() .map(id => String(id).trim()); if (existingIds.includes(orderId)) { return ContentService.createTextOutput("EXISTS") .setMimeType(ContentService.MimeType.TEXT); } } // ✅ 無商品 if (items.length === 0) { sheet.appendRow([ email, orderId, "已接收訂單", createdAt, "無商品", "", "", "" ]); } else { // ✅ 批次寫入 const rows = items.map(item => [ email, orderId, "已接收訂單", createdAt, item.title, item.quantity, item.price, item.image?.src || "" ]); sheet.getRange( sheet.getLastRow() + 1, 1, rows.length, rows[0].length ).setValues(rows); } return ContentService.createTextOutput("OK") .setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput("Error: " + error.toString()) .setMimeType(ContentService.MimeType.TEXT); } }
